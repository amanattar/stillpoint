```swift
import Foundation
import GRDB

public struct SidebarProject: Sendable, Equatable, Identifiable {
    public var item: Item
    /// Closed to-dos ÷ all to-dos (0 when empty), for the progress pie.
    public var progress: Double

    public var id: String { item.id }
}

public struct SidebarArea: Sendable, Equatable, Identifiable {
    public var area: Area
    public var projects: [SidebarProject]

    public var id: String { area.id }
}

/// Everything the sidebar shows below the fixed lists.
public struct SidebarContent: Sendable, Equatable {
    /// Open to-dos in the Inbox.
    public var inboxCount: Int
    /// Open to-dos and projects in Today.
    public var todayCount: Int
    /// Open Today items whose deadline is today or earlier (shown as a red badge).
    public var dueCount: Int
    /// Active projects that aren't in an area.
    public var looseProjects: [SidebarProject]
    public var areas: [SidebarArea]
}

extension ListQueries {
    func sidebar() throws -> SidebarContent {
        let inboxCount = try Int.fetchOne(
            db,
            sql: "SELECT COUNT(*) FROM item WHERE item.kind = 0 AND item.bucket = 0 AND item.status = 0 AND \(visible)",
            arguments: visibleArguments
        ) ?? 0
        let todayCondition = """
            item.kind IN (0, 1) AND item.status = 0 AND \(visible)
              AND ((item.bucket = 1 AND item.startDay <= ?) OR item.deadlineDay <= ?)
            """
        let todayCount = try Int.fetchOne(
            db, sql: "SELECT COUNT(*) FROM item WHERE \(todayCondition)",
            arguments: visibleArguments + [today, today]
        ) ?? 0
        let dueCount = try Int.fetchOne(
            db, sql: "SELECT COUNT(*) FROM item WHERE \(todayCondition) AND item.deadlineDay <= ?",
            arguments: visibleArguments + [today, today, today]
        ) ?? 0

        let projects = try Item.fetchAll(
            db,
            sql: """
            SELECT item.* FROM item WHERE item.kind = 1 AND \(visible)
              AND item.bucket = 1 AND (item.startDay IS NULL OR item.startDay <= ?)
            ORDER BY item.sortKey
            """,
            arguments: visibleArguments + [today]
        )
        let progress = try projectProgress(for: projects.map(\.id))
        let rows = projects.map { SidebarProject(item: $0, progress: progress[$0.id] ?? 0) }
        let areas = try Area.filter(Column("deletedAt") == nil).order(Column("sortKey")).fetchAll(db)
        return SidebarContent(
            inboxCount: inboxCount,
            todayCount: todayCount,
            dueCount: dueCount,
            looseProjects: rows.filter { $0.item.areaID == nil },
            areas: areas.map { area in SidebarArea(area: area, projects: rows.filter { $0.item.areaID == area.id }) }
        )
    }

    /// Closed ÷ total live to-dos for each project.
    func projectProgress(for projectIDs: [String]) throws -> [String: Double] {
        guard !projectIDs.isEmpty else { return [:] }
        let ids = String(decoding: try JSONEncoder().encode(projectIDs), as: UTF8.self)
        let rows = try Row.fetchAll(
            db,
            sql: """
            SELECT projectID, COUNT(*) AS total, SUM(status != 0) AS closed FROM item
            WHERE kind = 0 AND deletedAt IS NULL AND trashedAt IS NULL
              AND projectID IN (SELECT value FROM json_each(?))
            GROUP BY projectID
            """,
            arguments: [ids]
        )
        var result: [String: Double] = [:]
        for row in rows {
            let total: Int = row["total"]
            let closed: Int = row["closed"]
            result[row["projectID"]] = total == 0 ? 0 : Double(closed) / Double(total)
        }
        return result
    }
}

extension Store {
    public func fetchSidebar() async throws -> SidebarContent {
        let env = self.env
        return try await read { db in try ListQueries(db: db, env: env).sidebar() }
    }

    public func observeSidebar() -> AsyncValueObservation<SidebarContent> {
        let env = self.env
        return ValueObservation
            .tracking { db in try ListQueries(db: db, env: env).sidebar() }
            .removeDuplicates()
            .values(in: writer)
    }

    /// Live settings (defaults until the first change is saved).
    public func observeSettings() -> AsyncValueObservation<Settings> {
        let env = self.env
        return ValueObservation
            .tracking { db in try Store.settings(in: db, now: env.now()) }
            .removeDuplicates()
            .values(in: writer)
    }
}
```

```swift
    case area(String)

    case plain
    case evening
    case day(Day)
    case restOfMonth(from: Day, through: Day)
    /// Any day in the month; the section covers the whole month.

    case area(String)
    case project(String)
    case heading(String)
```

```swift
import Foundation
import GRDB

/// Builds `ListContent` for each list (spec §4.2).
struct ListQueries {
    let db: Database
    let env: ModelEnvironment
    let today: Day
    let settings: Settings
    /// Not tombstoned, not trashed, not logged. Bind `visibleArguments` after it.
    let visible: String
    let visibleArguments: StatementArguments
    let logged: String
    let loggedArguments: StatementArguments

    init(db: Database, env: ModelEnvironment) throws {
        self.db = db
        self.env = env
        let now = env.now()
        today = Day(now, calendar: env.calendar)
        settings = try Store.settings(in: db, now: now)

        let cutoff = settings.manualLogCutoff?.timeIntervalSince1970 ?? -Double.greatestFiniteMagnitude
        let policy: String
        var policyArguments: StatementArguments = []
        switch settings.logPolicy {
        case .immediately:
            policy = "1"
        case .daily:
            policy = "item.stopDate < ?"
            policyArguments = [today.startOfDay(in: env.calendar).timeIntervalSince1970]
        case .manually:
            policy = "0"
        }
        logged = "(item.status != 0 AND item.stopDate IS NOT NULL AND (item.stopDate <= ? OR \(policy)))"
        loggedArguments = [cutoff] + policyArguments
        visible = "(item.deletedAt IS NULL AND item.trashedAt IS NULL AND NOT \(logged))"
        visibleArguments = loggedArguments
    }

    func fetch(_ list: ListID, tagFilter: String?) throws -> ListContent {
        var content: ListContent
        switch list {
        case .inbox: content = try inbox()
        case .today: content = try todayList()
        case .upcoming: content = try upcoming()
        case .anytime: content = try anytime()
        case .someday: content = try someday()
        case .deadlines: content = try deadlines()
        case .logbook: content = try logbook()
        case .trash: content = try trash()
        case .allProjects: content = try projects(later: false)
        case .laterProjects: content = try projects(later: true)
        case .project(let id): content = try project(id)
        case .area(let id): content = try area(id)
        }
        if let tagFilter {
            content = try filter(content, byTag: tagFilter)
        }
        let everything = content.allItems + content.later + content.logged
        content.tagIDsByItem = try tagIDs(for: everything)
        let tagIDs = Set(content.tagIDsByItem.values.joined())
        content.tagTitles = Dictionary(uniqueKeysWithValues: try Tag.fetchAll(db, keys: tagIDs).map { ($0.id, $0.title) })
        content.parentTitles = try parentTitles(for: everything)
        content.checklists = try checklists(for: everything.filter { $0.kind == .todo })
        content.projectProgress = try projectProgress(for: everything.filter { $0.kind == .project }.map(\.id))
        return content
    }

    // MARK: Lists

    private func inbox() throws -> ListContent {
        let items = try select("item.kind = 0 AND item.bucket = 0 AND \(visible)", visibleArguments)
        return single(id: "inbox", items)
    }

    private func todayList() throws -> ListContent {
        let items = try select(
            """
            item.kind IN (0, 1) AND \(visible) AND (
              (item.bucket = 1 AND item.startDay <= ?) OR item.deadlineDay <= ?)
            """,
            visibleArguments + [today, today],
            order: "item.todaySortKey IS NULL, item.todaySortKey, item.startDay, item.sortKey"
        )
        let main = items.filter { !$0.isEvening }
        let evening = items.filter(\.isEvening)
        var sections: [ListSection] = []
        if settings.groupTodayByProject {
            sections += try groupedByParent(main, idPrefix: "today")
        } else if !main.isEmpty {
            sections.append(ListSection(id: "today", kind: .plain, title: nil, items: main))
        }
        if !evening.isEmpty {
            sections.append(ListSection(id: "evening", kind: .evening, title: "This Evening", items: evening))
        }
        return ListContent(sections: sections, later: [], logged: [], tagIDsByItem: [:])
    }

    private func upcoming() throws -> ListContent {
        let items = try select(
            """
            item.kind IN (0, 1) AND \(visible) AND (
              (item.bucket = 1 AND item.startDay > ?) OR item.deadlineDay > ?)
            """,
            visibleArguments + [today, today]
        )
        // Each item appears once: on its start day if that's in the future, else its deadline.
        let dated = items.map { item -> (Day, Item) in
            if let start = item.startDay, start > today { return (start, item) }
            return (item.deadlineDay!, item)
        }.sorted { ($0.0, $0.1.sortKey) < ($1.0, $1.1.sortKey) }

        let calendar = env.calendar
        var sections: [ListSection] = []
        // The next seven days always appear, even when empty.
        for offset in 1...7 {
            let day = today.adding(days: offset, calendar: calendar)
            sections.append(ListSection(id: "day.\(day.rawValue)", kind: .day(day), title: nil, items: dated.filter { $0.0 == day }.map(\.1)))
        }
        let weekEnd = today.adding(days: 7, calendar: calendar)
        let monthEnd = Day(year: weekEnd.year, month: weekEnd.month,
                           day: calendar.range(of: .day, in: .month, for: weekEnd.date(in: calendar))!.count)
        if weekEnd < monthEnd {
            let from = weekEnd.adding(days: 1, calendar: calendar)
            let rest = dated.filter { $0.0 >= from && $0.0 <= monthEnd }.map(\.1)
            sections.append(ListSection(id: "rest.\(from.rawValue)", kind: .restOfMonth(from: from, through: monthEnd), title: nil, items: rest))
        }
        let later = dated.filter { $0.0 > monthEnd }
        var months: [Int: [Item]] = [:]
        var monthOrder: [Int] = []
        for (day, item) in later {
            let key = day.year * 100 + day.month
            if months[key] == nil { monthOrder.append(key) }
            months[key, default: []].append(item)
        }
        for key in monthOrder {
            let first = Day(year: key / 100, month: key % 100, day: 1)
            sections.append(ListSection(id: "month.\(key)", kind: .month(first), title: nil, items: months[key]!))
        }
        return ListContent(sections: sections, later: [], logged: [], tagIDsByItem: [:])
    }

    private func anytime() throws -> ListContent {
        let todos = try select(
            """
            item.kind = 0 AND \(visible) AND item.bucket = 1 AND (item.startDay IS NULL OR item.startDay <= ?)
              AND (item.projectID IS NULL OR EXISTS (
                SELECT 1 FROM item p WHERE p.id = item.projectID AND p.bucket = 1
                  AND (p.startDay IS NULL OR p.startDay <= ?)))
            """,
            visibleArguments + [today, today]
        )
        return ListContent(sections: try groupedByAreaAndProject(todos), later: [], logged: [], tagIDsByItem: [:])
    }

    private func someday() throws -> ListContent {
        let items = try select(
            """
            item.kind IN (0, 1) AND \(visible) AND item.bucket = 2 AND item.startDay IS NULL
              AND (item.projectID IS NULL OR NOT EXISTS (
                SELECT 1 FROM item p WHERE p.id = item.projectID AND p.bucket = 2))
            """,
            visibleArguments
        )
        return ListContent(sections: try groupedByAreaAndProject(items), later: [], logged: [], tagIDsByItem: [:])
    }

    private func deadlines() throws -> ListContent {
        let items = try select(
            "item.kind IN (0, 1) AND \(visible) AND item.status = 0 AND item.deadlineDay IS NOT NULL",
            visibleArguments,
            order: "item.deadlineDay, item.sortKey"
        )
        var sections: [ListSection] = []
        for item in items {
            let day = item.deadlineDay!
            if sections.last?.kind == .day(day) {
                sections[sections.count - 1].items.append(item)
            } else {
                sections.append(ListSection(id: "deadline.\(day.rawValue)", kind: .day(day), title: nil, items: [item]))
            }
        }
        return ListContent(sections: sections, later: [], logged: [], tagIDsByItem: [:])
    }

    private func logbook() throws -> ListContent {
        let items = try select(
            "item.kind IN (0, 1) AND item.deletedAt IS NULL AND item.trashedAt IS NULL AND \(logged)",
            loggedArguments,
            order: "item.stopDate DESC"
        )
        let loggedProjects = Set(items.filter { $0.kind == .project }.map(\.id))
        var sections: [ListSection] = []
        for item in items where !(item.projectID.map(loggedProjects.contains) ?? false) {
            let day = Day(item.stopDate!, calendar: env.calendar)
            if sections.last?.kind == .day(day) {
                sections[sections.count - 1].items.append(item)
            } else {
                sections.append(ListSection(id: "logged.\(day.rawValue)", kind: .day(day), title: nil, items: [item]))
            }
        }
        return ListContent(sections: sections, later: [], logged: [], tagIDsByItem: [:])
    }

    private func trash() throws -> ListContent {
        let items = try select(
            """
            item.deletedAt IS NULL AND item.trashedAt IS NOT NULL AND NOT EXISTS (
              SELECT 1 FROM item p WHERE p.id IN (item.projectID, item.headingID) AND p.trashedAt IS NOT NULL)
            """,
            [],
            order: "item.trashedAt DESC, item.sortKey"
        )
        return single(id: "trash", items)
    }

    private func projects(later: Bool) throws -> ListContent {
        let condition = later
            ? "(item.bucket = 2 OR item.startDay > ?)"
            : "item.bucket = 1 AND (item.startDay IS NULL OR item.startDay <= ?)"
        let projects = try select("item.kind = 1 AND \(visible) AND \(condition)", visibleArguments + [today])
        var sections: [ListSection] = []
        let loose = projects.filter { $0.areaID == nil }
        if !loose.isEmpty {
            sections.append(ListSection(id: "projects.none", kind: .plain, title: nil, items: loose))
        }
        for area in try liveAreas() {
            let inArea = projects.filter { $0.areaID == area.id }
            if !inArea.isEmpty {
                sections.append(ListSection(id: "projects.\(area.id)", kind: .area(area.id), title: area.title, items: inArea))
            }
        }
        return ListContent(sections: sections, later: [], logged: [], tagIDsByItem: [:])
    }

    private func project(_ projectID: String) throws -> ListContent {
        let children = try select(
            "item.projectID = ? AND item.deletedAt IS NULL AND item.trashedAt IS NULL",
            [projectID]
        )
        let isLogged = try loggedIDs(children.map(\.id))
        let headings = children.filter { $0.kind == .heading }
        let archivedHeadings = Set(headings.filter { $0.status != .open }.map(\.id))
        let todos = children.filter { $0.kind == .todo }

        var logged: [Item] = []
        var later: [Item] = []
        var current: [Item] = []
        for todo in todos {
            if isLogged.contains(todo.id) || (todo.headingID.map(archivedHeadings.contains) ?? false) {
                logged.append(todo)
            } else if isLater(todo) {
                later.append(todo)
            } else {
                current.append(todo)
            }
        }

        var sections: [ListSection] = []
        let loose = current.filter { $0.headingID == nil }
        if !loose.isEmpty {
            sections.append(ListSection(id: "project.\(projectID)", kind: .plain, title: nil, items: loose))
        }
        for heading in headings where heading.status == .open {
            sections.append(ListSection(id: "heading.\(heading.id)", kind: .heading(heading.id), title: heading.title,
                                        items: current.filter { $0.headingID == heading.id }))
        }
        logged.sort { ($0.stopDate ?? .distantPast) > ($1.stopDate ?? .distantPast) }
        return ListContent(sections: sections, later: later, logged: logged, tagIDsByItem: [:])
    }

    private func area(_ areaID: String) throws -> ListContent {
        let children = try select(
            "item.areaID = ? AND item.projectID IS NULL AND item.kind IN (0, 1) AND item.deletedAt IS NULL AND item.trashedAt IS NULL",
            [areaID]
        )
        let isLogged = try loggedIDs(children.map(\.id))
        var logged: [Item] = []
        var later: [Item] = []
        var current: [Item] = []
        for item in children {
            if isLogged.contains(item.id) {
                logged.append(item)
            } else if isLater(item) {
                later.append(item)
            } else {
                current.append(item)
            }
        }
        var sections: [ListSection] = []
        let todos = current.filter { $0.kind == .todo }
        let projects = current.filter { $0.kind == .project }
        if !todos.isEmpty {
            sections.append(ListSection(id: "area.\(areaID).todos", kind: .plain, title: nil, items: todos))
        }
        if !projects.isEmpty {
            sections.append(ListSection(id: "area.\(areaID).projects", kind: .plain, title: nil, items: projects))
        }
        logged.sort { ($0.stopDate ?? .distantPast) > ($1.stopDate ?? .distantPast) }
        return ListContent(sections: sections, later: later, logged: logged, tagIDsByItem: [:])
    }

    // MARK: Grouping

    /// Loose items first, then each area's loose items and projects, then projects without an area.
    private func groupedByAreaAndProject(_ items: [Item]) throws -> [ListSection] {
        var loose: [Item] = []
        var byArea: [String: [Item]] = [:]
        var byProject: [String: [Item]] = [:]
        for item in items {
            if let projectID = item.projectID {
                byProject[projectID, default: []].append(item)
            } else if let areaID = item.areaID {
                byArea[areaID, default: []].append(item)
            } else {
                loose.append(item)
            }
        }
        let parents = try Item.fetchAll(db, keys: Set(byProject.keys)).sorted { $0.sortKey < $1.sortKey }
        let headingKeys = try headingSortKeys(projectIDs: Set(byProject.keys))

        func projectSections(areaID: String?) -> [ListSection] {
            parents.filter { $0.areaID == areaID }.map { project in
                // Within a project: no-heading items first, then by heading order, then by item order.
                let members = byProject[project.id]!
                    .map { (headingKeys[$0.headingID ?? ""] ?? "", $0) }
                    .sorted { ($0.0, $0.1.sortKey) < ($1.0, $1.1.sortKey) }
                    .map(\.1)
                return ListSection(id: "project.\(project.id)", kind: .project(project.id), title: project.title, items: members)
            }
        }

        var sections: [ListSection] = []
        if !loose.isEmpty {
            sections.append(ListSection(id: "none", kind: .plain, title: nil, items: loose))
        }
        for area in try liveAreas() {
            if let areaItems = byArea[area.id] {
                sections.append(ListSection(id: "area.\(area.id)", kind: .area(area.id), title: area.title, items: areaItems))
            }
            sections += projectSections(areaID: area.id)
        }
        sections += projectSections(areaID: nil)
        return sections
    }

    /// Today's "group by project" option: loose items first, then one section per project or area.
    private func groupedByParent(_ items: [Item], idPrefix: String) throws -> [ListSection] {
        var order: [String] = []
        var groups: [String: [Item]] = [:]
        for item in items {
            let key = item.projectID ?? item.areaID ?? ""
            if groups[key] == nil { order.append(key) }
            groups[key, default: []].append(item)
        }
        order.sort { $0.isEmpty && !$1.isEmpty }
        return try order.map { key in
            if key.isEmpty {
                return ListSection(id: idPrefix, kind: .plain, title: nil, items: groups[key]!)
            }
            if let area = try Area.fetchOne(db, key: key) {
                return ListSection(id: "\(idPrefix).\(key)", kind: .area(key), title: area.title, items: groups[key]!)
            }
            let project = try Item.fetchOne(db, key: key)
            return ListSection(id: "\(idPrefix).\(key)", kind: .project(key), title: project?.title, items: groups[key]!)
        }
    }

    // MARK: Tags

    /// Keeps items tagged with `tagID` or any tag nested under it, directly or
    /// through their project or area.
    private func filter(_ content: ListContent, byTag tagID: String) throws -> ListContent {
        var wanted: Set<String> = [tagID]
        var frontier = [tagID]
        while let current = frontier.popLast() {
            let children = try String.fetchAll(db, sql: "SELECT id FROM tag WHERE parentID = ? AND deletedAt IS NULL", arguments: [current])
            for child in children where wanted.insert(child).inserted {
                frontier.append(child)
            }
        }
        let tagged = Set(try String.fetchAll(
            db,
            sql: "SELECT itemID FROM item_tag WHERE deletedAt IS NULL AND tagID IN (SELECT value FROM json_each(?))",
            arguments: [jsonArray(wanted)]
        ))
        let taggedAreas = Set(try String.fetchAll(
            db,
            sql: "SELECT areaID FROM area_tag WHERE deletedAt IS NULL AND tagID IN (SELECT value FROM json_each(?))",
            arguments: [jsonArray(wanted)]
        ))
        var projectAreas: [String: String] = [:]
        for project in try Item.fetchAll(db, keys: Set((content.allItems + content.later + content.logged).compactMap(\.projectID))) {
            projectAreas[project.id] = project.areaID
        }
        func matches(_ item: Item) -> Bool {
            if tagged.contains(item.id) { return true }
            if let projectID = item.projectID {
                if tagged.contains(projectID) { return true }
                if let areaID = projectAreas[projectID], taggedAreas.contains(areaID) { return true }
            }
            if let areaID = item.areaID, taggedAreas.contains(areaID) { return true }
            return false
        }
        var result = content
        result.sections = content.sections.compactMap { section in
            var section = section
            section.items = section.items.filter(matches)
            // Day sections in Upcoming stay even when empty, like the unfiltered list.
            if case .day = section.kind { return section }
            return section.items.isEmpty ? nil : section
        }
        result.later = content.later.filter(matches)
        result.logged = content.logged.filter(matches)
        return result
    }

    private func tagIDs(for items: [Item]) throws -> [String: [String]] {
        guard !items.isEmpty else { return [:] }
        let rows = try Row.fetchAll(
            db,
            sql: """
            SELECT item_tag.itemID, item_tag.tagID FROM item_tag JOIN tag ON tag.id = item_tag.tagID
            WHERE item_tag.deletedAt IS NULL AND tag.deletedAt IS NULL
              AND item_tag.itemID IN (SELECT value FROM json_each(?))
            ORDER BY tag.sortKey
            """,
            arguments: [jsonArray(Set(items.map(\.id)))]
        )
        var result: [String: [String]] = [:]
        for row in rows {
            result[row["itemID"], default: []].append(row["tagID"])
        }
        return result
    }

    // MARK: Row details

    private func parentTitles(for items: [Item]) throws -> [String: String] {
        let projectIDs = Set(items.compactMap(\.projectID))
        let areaIDs = Set(items.compactMap(\.areaID))
        var titles: [String: String] = [:]
        for project in try Item.fetchAll(db, keys: projectIDs) { titles[project.id] = project.title }
        for area in try Area.fetchAll(db, keys: areaIDs) { titles[area.id] = area.title }
        var result: [String: String] = [:]
        for item in items {
            if let parent = (item.projectID ?? item.areaID).flatMap({ titles[$0] }) {
                result[item.id] = parent
            }
        }
        return result
    }

    private func checklists(for todos: [Item]) throws -> [String: ChecklistProgress] {
        guard !todos.isEmpty else { return [:] }
        let rows = try Row.fetchAll(
            db,
            sql: """
            SELECT itemID, COUNT(*) AS total, SUM(isDone) AS done FROM checklist_item
            WHERE deletedAt IS NULL AND itemID IN (SELECT value FROM json_each(?))
            GROUP BY itemID
            """,
            arguments: [jsonArray(Set(todos.map(\.id)))]
        )
        var result: [String: ChecklistProgress] = [:]
        for row in rows {
            result[row["itemID"]] = ChecklistProgress(done: row["done"], total: row["total"])
        }
        return result
    }

    // MARK: Helpers

    private func select(_ condition: String, _ arguments: StatementArguments, order: String = "item.sortKey") throws -> [Item] {
        try Item.fetchAll(db, sql: "SELECT item.* FROM item WHERE \(condition) ORDER BY \(order)", arguments: arguments)
    }

    private func single(id: String, _ items: [Item]) -> ListContent {
        ListContent(sections: items.isEmpty ? [] : [ListSection(id: id, kind: .plain, title: nil, items: items)],
                    later: [], logged: [], tagIDsByItem: [:])
    }

    private func isLater(_ item: Item) -> Bool {
        item.bucket == .someday || (item.startDay.map { $0 > today } ?? false)
    }

    private func loggedIDs(_ ids: [String]) throws -> Set<String> {
        guard !ids.isEmpty else { return [] }
        return Set(try String.fetchAll(
            db,
            sql: "SELECT item.id FROM item WHERE item.id IN (SELECT value FROM json_each(?)) AND \(logged)",
            arguments: [jsonArray(Set(ids))] + loggedArguments
        ))
    }

    private func liveAreas() throws -> [Area] {
        try Area.filter(Column("deletedAt") == nil).order(Column("sortKey")).fetchAll(db)
    }

    private func headingSortKeys(projectIDs: Set<String>) throws -> [String: String] {
        guard !projectIDs.isEmpty else { return [:] }
        let rows = try Row.fetchAll(
            db,
            sql: "SELECT id, sortKey FROM item WHERE kind = 2 AND projectID IN (SELECT value FROM json_each(?))",
            arguments: [jsonArray(projectIDs)]
        )
        return Dictionary(uniqueKeysWithValues: rows.map { ($0["id"], $0["sortKey"]) })
    }

    private func jsonArray(_ ids: Set<String>) -> String {
        String(decoding: try! JSONEncoder().encode(ids.sorted()), as: UTF8.self)
    }
```

```swift
import SwiftUI

/// A colour defined in code, so tests can check contrast without rendering.
public struct RGBA: Hashable, Sendable {
    public var red: Double
    public var green: Double
    public var blue: Double
    public var alpha: Double

    /// `0xRRGGBB`, fully opaque.
    public init(hex: UInt32, alpha: Double = 1) {
        red = Double((hex >> 16) & 0xFF) / 255
        green = Double((hex >> 8) & 0xFF) / 255
        blue = Double(hex & 0xFF) / 255
        self.alpha = alpha
    }

    public var color: Color {
        Color(.sRGB, red: red, green: green, blue: blue, opacity: alpha)
    }

    /// WCAG relative luminance (alpha ignored).
    public var luminance: Double {
        func channel(_ c: Double) -> Double {
            c <= 0.039_28 ? c / 12.92 : pow((c + 0.055) / 1.055, 2.4)
        }
        return 0.2126 * channel(red) + 0.7152 * channel(green) + 0.0722 * channel(blue)
    }

    /// WCAG contrast ratio, from 1 (none) to 21 (black on white).
    public func contrast(with other: RGBA) -> Double {
        let (a, b) = (luminance + 0.05, other.luminance + 0.05)
        return max(a, b) / min(a, b)
    }
}
```

```swift
import SwiftUI

/// A project's progress: an outlined circle that fills clockwise from the top.
public struct ProgressPie: View {
    public var progress: Double
    public var tint: Color?
    @Environment(\.palette) private var palette
    @Environment(\.metrics) private var metrics

    public init(_ progress: Double, tint: Color? = nil) {
        self.progress = min(max(progress, 0), 1)
        self.tint = tint
    }

    public var body: some View {
        let side = metrics.checkbox
        let color = tint ?? palette.inbox.color
        ZStack {
            Circle().strokeBorder(color, lineWidth: max(1.2, side / 10))
            PieSlice(fraction: progress)
                .fill(color)
                .padding(side * 0.18)
        }
        .frame(width: side, height: side)
        .accessibilityElement()
        .accessibilityLabel(Text("\(Int((progress * 100).rounded())) percent complete"))
    }
}

struct PieSlice: Shape {
    var fraction: Double

    func path(in rect: CGRect) -> Path {
        guard fraction > 0 else { return Path() }
        if fraction >= 1 { return Path(ellipseIn: rect) }
        let center = CGPoint(x: rect.midX, y: rect.midY)
        var path = Path()
        path.move(to: center)
        path.addArc(center: center, radius: rect.width / 2,
                    startAngle: .degrees(-90), endAngle: .degrees(-90 + 360 * fraction), clockwise: false)
        path.closeSubpath()
        return path
    }
}
```

```swift
import Foundation
import StillpointModel

/// Human wording for days: section headers and deadline badges.
public struct DateText: Sendable {
    public let today: Day
    public let calendar: Calendar
    public let locale: Locale

    public init(today: Day, calendar: Calendar, locale: Locale = .autoupdatingCurrent) {
        self.today = today
        self.calendar = calendar
        self.locale = locale
    }

    /// Whole days from today to `day` (negative = past).
    public func offset(to day: Day) -> Int {
        calendar.dateComponents([.day], from: today.date(in: calendar), to: day.date(in: calendar)).day ?? 0
    }

    public func format(_ day: Day, _ template: String) -> String {
        let formatter = DateFormatter()
        formatter.calendar = calendar
        formatter.timeZone = calendar.timeZone
        formatter.locale = locale
        formatter.setLocalizedDateFormatFromTemplate(template)
        return formatter.string(from: day.date(in: calendar))
    }

    /// "Today", "Tomorrow", "Yesterday", a weekday within the coming week,
    /// otherwise a short date (with the year when it isn't this year).
    public func dayTitle(_ day: Day) -> String {
        switch offset(to: day) {
        case 0: return "Today"
        case 1: return "Tomorrow"
        case -1: return "Yesterday"
        case 2...7: return format(day, "EEEE")
        default: return format(day, day.year == today.year ? "EEEMMMd" : "EEEMMMdy")
        }
    }

    /// Day sections 1–7 days ahead show the day number large beside their title.
    public func showsDayNumber(_ day: Day) -> Bool {
        (1...7).contains(offset(to: day))
    }

    public func sectionTitle(_ section: ListSection) -> String? {
        switch section.kind {
        case .plain, .area, .project, .heading:
            return section.title
        case .evening:
            return "This Evening"
        case .day(let day):
            return dayTitle(day)
        case .restOfMonth(let from, let through):
            return "\(format(from, "MMMd"))–\(through.day)"
        case .month(let day):
            return format(day, day.year == today.year ? "MMMM" : "MMMMy")
        }
    }

    /// Deadline badge text and whether it's due (today or overdue).
    public func deadline(_ day: Day) -> (text: String, isDue: Bool) {
        let days = offset(to: day)
        let text: String = switch days {
        case 0: "Today"
        case 1: "Tomorrow"
        case -1: "Yesterday"
        case ..<(-1): "\(-days) days ago"
        case 2...6: format(day, "EEE")
        default: format(day, "MMMd")
        }
        return (text, days <= 0)
    }
}
```

```swift
import StillpointModel

/// Everything one row displays, derived from the item and its list.
public struct RowModel: Identifiable, Equatable {
    public enum Leading: Equatable {
        case checkbox(ItemStatus)
        case progress(Double)
    }

    public var id: String
    public var title: String
    public var leading: Leading
    /// Project or area name, hidden when the row is already inside that project or area.
    public var detail: String?
    /// Amber sun: a to-do that's in Today, shown in lists other than Today.
    public var showsTodayMark: Bool
    /// Indigo moon: scheduled for this evening, shown in lists other than Today.
    public var showsEveningMark: Bool
    public var hasNotes: Bool
    public var checklist: ChecklistProgress?
    public var tags: [String]
    public var deadline: (text: String, isDue: Bool)?
    public var isClosed: Bool

    public static func == (lhs: RowModel, rhs: RowModel) -> Bool {
        lhs.id == rhs.id && lhs.title == rhs.title && lhs.leading == rhs.leading && lhs.detail == rhs.detail
            && lhs.showsTodayMark == rhs.showsTodayMark && lhs.showsEveningMark == rhs.showsEveningMark
            && lhs.hasNotes == rhs.hasNotes && lhs.checklist == rhs.checklist && lhs.tags == rhs.tags
            && lhs.deadline?.text == rhs.deadline?.text && lhs.deadline?.isDue == rhs.deadline?.isDue
            && lhs.isClosed == rhs.isClosed
    }

    public init(item: Item, in list: ListID, content: ListContent, dates: DateText) {
        id = item.id
        title = item.title
        leading = item.kind == .project ? .progress(content.projectProgress[item.id] ?? 0) : .checkbox(item.status)

        let parentID = item.projectID ?? item.areaID
        let insideParent: Bool = switch list {
        case .project(let id), .area(let id): parentID == id
        default: false
        }
        detail = insideParent ? nil : content.parentTitles[item.id]

        let isInToday = item.bucket == .anytime && (item.startDay.map { $0 <= dates.today } ?? false)
        showsEveningMark = list != .today && isInToday && item.isEvening
        showsTodayMark = list != .today && isInToday && !item.isEvening
        hasNotes = !item.notes.isEmpty
        checklist = content.checklists[item.id]
        tags = (content.tagIDsByItem[item.id] ?? []).compactMap { content.tagTitles[$0] }
        deadline = item.deadlineDay.map { dates.deadline($0) }
        isClosed = item.status != .open
    }
}
```

```swift
/// A list's scrolling content: header, sections, rows and the later/logged disclosures.

public struct ListBody: View {

    let title: String
    let notes: String
    let content: ListContent?
    let dates: DateText
    @Binding var showsLater: Bool
    @Binding var showsLogged: Bool

    @Environment(\.metrics) private var metrics

    public init(
        list: ListID, title: String, notes: String, content: ListContent?, dates: DateText,
        showsLater: Binding<Bool>, showsLogged: Binding<Bool>
    ) {
        self.list = list
        self.title = title
        self.notes = notes
        self.content = content
        self.dates = dates
        _showsLater = showsLater
        _showsLogged = showsLogged
    }

    public var body: some View {
        LazyVStack(alignment: .leading, spacing: 0) {
            ListHeader(symbol: list.symbol, title: title, notes: notes)
            if let content {
                if content.sections.isEmpty, content.later.isEmpty, content.logged.isEmpty {
                    EmptyListView(symbol: list.symbol)
                }
                ForEach(content.sections) { section in
                    SectionHeader(section: section, title: dates.sectionTitle(section), dates: dates)
                    ForEach(section.items) { item in
                        TaskRow(row: RowModel(item: item, in: list, content: content, dates: dates))
                    }
                }
                disclosure(content.later, label: "later", isOn: $showsLater, content: content)
                disclosure(content.logged, label: "logged", isOn: $showsLogged, content: content)
            }
        }
    }

    @ViewBuilder
    private func disclosure(_ items: [Item], label: String, isOn: Binding<Bool>, content: ListContent) -> some View {
        if !items.isEmpty {

            }
        }
    }
```

```swift
import Foundation
import Observation
import StillpointModel

/// Publishes the current calendar day. Lists re-subscribe when it changes,
/// so Today moves forward at local midnight, after waking from sleep on a
/// new day, and when the time zone changes.
@MainActor
@Observable
final class DayClock {
    private(set) var today: Day
    let calendar: Calendar
    @ObservationIgnored var onDayChange: (() -> Void)?
    @ObservationIgnored private let now: @Sendable () -> Date
    @ObservationIgnored private var observers: [any NSObjectProtocol] = []

    init(
        calendar: Calendar = .autoupdatingCurrent,
        now: @escaping @Sendable () -> Date = { Date() },
        center: NotificationCenter = .default
    ) {
        self.calendar = calendar
        self.now = now
        today = Day(now(), calendar: calendar)
        for name in [Notification.Name.NSCalendarDayChanged, .NSSystemTimeZoneDidChange] {
            observers.append(center.addObserver(forName: name, object: nil, queue: .main) { [weak self] _ in
                MainActor.assumeIsolated { self?.refresh() }
            })
        }
    }

    /// Re-reads the day; notifies only when it actually changed.
    func refresh() {
        let day = Day(now(), calendar: calendar)
        guard day != today else { return }
        today = day
        onDayChange?()
    }
}
```

`App/Startup/AppModel.swift`:

```swift
import Foundation
import GRDB
import Observation
import StillpointModel

/// Opens the database and owns the `Store` for every window.
@MainActor
@Observable
final class AppModel {
    struct DatabaseProblem: Equatable {
        var message: String
        /// The newest backup, offered for restore when the database was damaged.
        var backup: URL?
        /// True when a damaged file was moved aside, so starting with an empty database is safe.
        var canStartFresh: Bool
    }

    enum Phase: Equatable {
        case loading
        case ready
        case failed(DatabaseProblem)
    }

    private(set) var phase: Phase = .loading
    private(set) var store: Store?
    private(set) var settings: Settings
    let clock: DayClock
    let location: DatabaseLocation

    @ObservationIgnored private let deviceID: String
    @ObservationIgnored private let now: @Sendable () -> Date
    @ObservationIgnored private var pool: DatabasePool?
    @ObservationIgnored private var settingsTask: Task<Void, Never>?

    init(location: DatabaseLocation, deviceID: String, now: @escaping @Sendable () -> Date = { Date() }, clock: DayClock) {
        self.location = location
        self.deviceID = deviceID
        self.now = now
        self.clock = clock
        settings = Settings(createdAt: now())
        clock.onDayChange = { [weak self] in self?.backUpIfDue() }
    }

    static func live() -> AppModel {
        let defaults = UserDefaults.standard
        let deviceID: String
        if let saved = defaults.string(forKey: "deviceID") {
            deviceID = saved
        } else {
            deviceID = UUID().uuidString.lowercased()
            defaults.set(deviceID, forKey: "deviceID")
        }
        return AppModel(location: isTestHost ? .temporary() : .resolve(defaults), deviceID: deviceID, clock: DayClock())
    }

    /// When the app is only hosting unit tests, it must never open the real database.
    static var isTestHost: Bool {
        ProcessInfo.processInfo.environment["XCTestConfigurationFilePath"] != nil
    }

    /// Opens the database once; later calls (from extra windows) do nothing.
    func start() {
        guard phase == .loading, store == nil else { return }
        open()
    }

    func restoreBackup(_ backup: URL) {
        do {
            try location.backups.restore(backup, to: location.database)
            phase = .loading
            open()
        } catch {
            phase = .failed(DatabaseProblem(message: "The backup couldn't be restored: \(error.localizedDescription)", backup: nil, canStartFresh: true))
        }
    }

    /// After a damaged file was moved aside: start with an empty database.
    func startFresh() {
        phase = .loading
        open()
    }

    private func open() {
        do {
            let pool = try AppDatabase.open(at: location.database, backups: location.backups, now: now())
            self.pool = pool
            #if DEBUG
            let assertOnRuleViolation = true
            #else
            let assertOnRuleViolation = false
            #endif
            let env = ModelEnvironment(now: now, calendar: clock.calendar, deviceID: deviceID, assertOnRuleViolation: assertOnRuleViolation)
            store = Store(writer: pool, env: env)
            phase = .ready
            backUpIfDue()
            observeSettings()
        } catch DatabaseOpenError.corrupted(_, let newestBackup) {
            phase = .failed(DatabaseProblem(
                message: "Your Stillpoint database is damaged. It has been moved aside so nothing is lost.",
                backup: newestBackup, canStartFresh: true
            ))
        } catch DatabaseOpenError.migrationFailed(let message, _) {
            phase = .failed(DatabaseProblem(
                message: "Stillpoint couldn't update your database (\(message)). Your data was restored from the backup made just before the update.",
                backup: nil, canStartFresh: false
            ))
        } catch {
            phase = .failed(DatabaseProblem(message: error.localizedDescription, backup: nil, canStartFresh: false))
        }
    }

    /// Daily backup: at launch and whenever the day changes while running.
    private func backUpIfDue() {
        guard let pool else { return }
        do {
            if try location.backups.isDue(now: now()) {
                try location.backups.makeBackup(from: pool, at: now())
            }
        } catch {
            // A failed backup must never stop the app; the next day change retries.
        }
    }

    private func observeSettings() {
        guard let store else { return }
        settingsTask?.cancel()
        settingsTask = Task { [weak self] in
            do {
                for try await settings in store.observeSettings() {
                    self?.settings = settings
                }
            } catch {}
        }
    }
}
```

```swift
struct Themed<Content: View>: View {
    let settings: StillpointModel.Settings
    @ViewBuilder let content: () -> Content

    var body: some View {
        Palettized(appearance: settings.appearance, content: content)

/// How the Appearance setting maps to SwiftUI's colour scheme and Stillpoint's palette.
enum AppearanceMapping {
    static func colorScheme(for appearance: Appearance) -> ColorScheme? {
        switch appearance {
        case .automatic: nil
        case .light: .light
        case .dark, .black: .dark
        }
    }

    static func variant(for appearance: Appearance, effective scheme: ColorScheme) -> PaletteVariant {
        if appearance == .black { return .black }
        return scheme == .dark ? .dark : .light
    }
}

/// Reads the effective colour scheme (after `preferredColorScheme`) and picks the palette.
private struct Palettized<Content: View>: View {
    let appearance: Appearance
    @ViewBuilder let content: () -> Content
    @Environment(\.colorScheme) private var scheme

    var body: some View {
        content().environment(\.palette, .variant(AppearanceMapping.variant(for: appearance, effective: scheme)))
```

```swift
import StillpointFeatures
import StillpointModel
```

```swift
/// The main pane for one list: header, sections and rows, updating live.
struct ListScreen: View {
    let store: Store
    let list: ListID
    @Environment(DayClock.self) private var clock
    @Environment(\.palette) private var palette
    @Environment(\.metrics) private var metrics
    @State private var content: ListContent?
    @State private var title = ""
    @State private var notes = ""
    @State private var showsLater = false
    @State private var showsLogged = false

    private struct Subscription: Equatable {
        let list: ListID
        let today: Day
    }

    var body: some View {
        ScrollView {
            ListBody(
                list: list, title: title, notes: notes, content: content,
                dates: DateText(today: clock.today, calendar: clock.calendar),
                showsLater: $showsLater, showsLogged: $showsLogged
            )
            .padding(.horizontal, metrics.contentInset)
            .padding(.bottom, metrics.contentInset)
        }
        .background(palette.background.color)
        .task(id: Subscription(list: list, today: clock.today)) {
            await observe()
        }
    }

    private func observe() async {
        showsLater = false
        showsLogged = false
        content = nil
        do {
            for try await update in store.observe(list) {
                (title, notes) = try await store.header(for: list)
                content = update
            }
        } catch {
            content = nil
        }
    }
}
```

```swift
import Foundation

/// Launch with `-dataFolderName Demo -seedDemoData YES` to fill a separate demo database;
/// it never touches the real one.
enum DemoLaunch {
    @MainActor
    static func seedIfRequested(_ model: AppModel, defaults: UserDefaults = .standard) async {
        guard defaults.bool(forKey: "seedDemoData"), defaults.string(forKey: "dataFolderName") != nil else { return }
        model.start()
        guard let store = model.store,
              let sidebar = try? await store.fetchSidebar(),
              sidebar.inboxCount == 0, sidebar.areas.isEmpty
        else { return }
        try? await DemoData.seed(store)
    }
}
```
